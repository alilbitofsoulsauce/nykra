class Stats:
    __slots__ = ('size', 'minimum', 'median', 'maximum', 'mean')
    def __init__(self, size, minimum, median, maximum, mean):
        self.size = size
        self.minimum = minimum
        self.median = median
        self.maximum = maximum
        self.mean = mean
    def __repr__(self):
        return f"Stats(size={self.size},minimum={self.minimum},median={self.median},maximum={self.maximum},mean={self.mean})"
    def __eq__(self, other):
        return (
            isinstance(other, Stats)
            and self.size == other.size
            and self.minimum == other.minimum
            and self.median == other.median
            and self.maximum == other.maximum
            and self.mean == other.mean
        )

def _running_min_windowed(iterable, maxlen):
    d = deque(maxlen=maxlen)
    it = iter(iterable)
    for val in it:
        d.append(val)
        if len(d) == maxlen:
            break
    if not d:
        return
    current = min(d)
    yield current
    for val in it:
        d.append(val)
        if val < current:
            current = val
        yield current

def running_mean(iterable, *, maxlen=None):
    from itertools import accumulate, count
    from operator import truediv
    if maxlen is None:
        return map(truediv, accumulate(iterable), count(1))
    maxlen = index(maxlen)
    if maxlen <= 0:
        raise ValueError('Window size should be positive')
    from collections import deque
    d = deque(maxlen=maxlen)
    it = iter(iterable)
    for val in it:
        d.append(val)
        if len(d) == maxlen:
            break
    if not d:
        return
    total = sum(d)
    count = len(d)
    mean = total / count
    yield mean
    for val in it:
        d.append(val)
        total += val - d[0]
        mean = total / maxlen
        yield mean
